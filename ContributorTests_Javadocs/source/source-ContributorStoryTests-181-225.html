183 /** C04 negative: test each individual lock and the whole-lesson lock.
184 * Success means programmatic edits are denied, GUI editability/labels agree, and
185 * title content is unchanged. Contributor lock controls are hidden and denied.
186 * @throws Exception if persistence or evidence fails */
187 public static void testLockedEditsAndIndicators() throws Exception {
188 for (ModelLessons.LessonField field : ModelLessons.LessonField.values()) {
189 Lesson lesson = fixture();
190 ModelLessons.toggleLock(lesson.getId(), field, admin);
191 Lesson changed = ModelLessons.readLesson(lesson.getId(), owner);
192 switch (field) {
193 case TITLE -> changed.setTitle("Denied");
194 case PROBLEM -> changed.setProblem("Denied");
195 case SOLUTION -> changed.setSolution("Denied");
196 case EXPERIENCE -> changed.setExperience("Denied");
197 }
198 ViewLessons.currentUser = owner;
199 ViewLessons.showLesson(ModelLessons.readLesson(lesson.getId(), owner));
200 boolean[] editable = {ViewLessons.text_Title.isEditable(), ViewLessons.text_Problem.isEditable(),
201 ViewLessons.text_Solution.isEditable(), ViewLessons.text_Experience.isEditable()};
202 check("C04-N locked " + field, denied(() -> ModelLessons.updateLesson(changed, owner))
203 && !editable[field.ordinal()] && ViewLessons.fieldLabels[field.ordinal()].getText().contains("[locked]"));
204 }
205 Lesson whole = fixture();
206 ModelLessons.setLessonsLocked(List.of(whole.getId()), true, admin);
207 whole.setTitle("Denied whole edit");
208 check("C04-N whole lesson", denied(() -> ModelLessons.updateLesson(whole, owner))
209 && ModelLessons.readLesson(whole.getId(), owner).getTitle().equals("Story lesson"));
210 check("C04-N contributor cannot change locks", denied(() -> ModelLessons.toggleLock(whole.getId(), ModelLessons.LessonField.TITLE, owner))
211 && denied(() -> ModelLessons.setLessonsLocked(List.of(whole.getId()), false, owner)));
212 ViewLessons.displayLessons(ViewLessons.theStage, owner);
213 ViewLessons.showLesson(ModelLessons.readLesson(whole.getId(), owner));
214 check("C04-N admin controls hidden", !ViewLessons.lockButtons[0].isVisible() && !ViewLessons.button_LockSelected.isVisible());
215 capture("09-contributor-locked-fields");
216 }
